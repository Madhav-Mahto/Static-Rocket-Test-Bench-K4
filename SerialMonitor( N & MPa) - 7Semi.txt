#include "HX711.h"

#define DT  4
#define SCK 5

HX711 scale;

float calibrationFactor = 1000.875151;  // CHANGE THIS after calibration

const float GRAVITY = 0.000981;       // kg -> kN

bool measuring = false;
bool hasResult = false;

float currentForce = 0;
float lastForce = 0;
float maxForce = 0;

// Pressure transducer

const int pressurePin = A0;

const float VREF = 5.0;
const float VMIN = 0.5;
const float VMAX = 4.5;
const float PMAX = 1.2;     // MPa

// Store last measured values
float LP = 0;   // Last Pressure (MPa)
float MP = 0;   // Max Pressure (MPa)

// ================= MENU =================

void showMenu() {
  Serial.println();
  Serial.println("========== PRESSURE TEST ==========");
  Serial.println("1. START READING");
  Serial.println("2. STOP READING");
  Serial.println("3. SHOW LAST RESULT");
  Serial.println("4. TARE Load Cell");
  Serial.println("5. SHOW MENU");
  Serial.println("===================================");
  Serial.println("Enter option:");
}

// ================= SETUP =================

void setup() {
  Serial.begin(9600);

  // Load cell
  scale.begin(DT, SCK);
  scale.set_scale(calibrationFactor);

  Serial.println("Taring Load Cell...");
  scale.tare();
  Serial.println("Tare Complete");

  // Pressure transducer
  Serial.println("CHAMBER PRESSURE");

  // Menu
  showMenu();
}

// ================= MAIN LOOP =================

void loop() {

  // ---------- CHECK USER COMMAND ----------
  if (Serial.available() > 0) {

    char command = Serial.read();

    // START
    if (command == '1') {

      measuring = true;

      // Load cell
      maxForce = 0;
      Serial.println();
      Serial.println("READING STARTED");
      Serial.println();
      Serial.println("Time(ms) | Force(kN)");

      // Pressure transducer
      MP = 0;
      Serial.println();
      Serial.println("Pressure(MPa)");
    }

    // STOP
    else if (command == '2') {

      measuring = false;

      Serial.println();
      Serial.println("READING STOPPED");

      // Load cell
      Serial.print("Last Force: ");
      Serial.print(lastForce, 5);
      Serial.println(" kN");
      Serial.print("Maximum Force: ");
      Serial.print(maxForce, 5);
      Serial.println(" kN");

      // Pressure transducer
      Serial.print("Last Chamber Pressure: ");
      Serial.print(LP, 3);
      Serial.println(" MPa");
      Serial.print("Max. Chamber Pressure: ");
      Serial.print(MP, 3);
      Serial.println(" MPa");

      showMenu();
    }

    // SHOW LAST RESULT
    else if (command == '3') {

      Serial.println();
      Serial.println("========== LAST RESULT ==========");

      if (hasResult) {
        // Load cell
        Serial.print("Last Force: ");
        Serial.print(lastForce, 2);
        Serial.println(" kN");
        Serial.print("Maximum Force: ");
        Serial.print(maxForce, 2);
        Serial.println(" kN");

        // Pressure transducer
        Serial.print("Last Pressure: ");
        Serial.print(LP, 3);
        Serial.println(" MPa");
        Serial.print("Max. Chamber Pressure: ");
        Serial.print(MP, 3);
        Serial.println(" MPa");
      }
      else {
        Serial.println("No measurement available.");
      }

      Serial.println("=================================");
    }

    // -------- TARE --------
    else if (command == '4') {

      if (!measuring) {
        Serial.println("Taring...");
        scale.tare();
        Serial.println("Tare Complete");
      }
      else {
        Serial.println("STOP measurement before TARE.");
      }
    }

    // SHOW MENU
    else if (command == '5') {
      showMenu();
    }
  }

  // ---------- READINGS ----------
  if (measuring) {

    // ----- Load cell -----
    if (scale.is_ready()) {

      float massKg = scale.get_units(5);

      // Convert kg to Newton
      currentForce = massKg * GRAVITY;

      lastForce = currentForce;
      hasResult = true;

      if (currentForce > maxForce) {
        maxForce = currentForce;
      }

      Serial.print(millis());
      Serial.print(" | ");
      Serial.print(currentForce, 2);
      Serial.println(" kN");
    }
    else {
      Serial.println("HX711 not detected!");
    }

    // ----- Pressure transducer -----
    int adcValue = analogRead(pressurePin);

    // ADC -> Voltage
    float voltage = adcValue * (VREF / 1023.0);

    // Voltage -> Pressure
    float pressureMPa = ((voltage - VMIN) / (VMAX - VMIN)) * PMAX;

    if (pressureMPa < 0) {
      pressureMPa = 0;
    }

    LP = pressureMPa;
    hasResult = true;

    if (pressureMPa > MP) {
      MP = pressureMPa;
    }

    Serial.print(pressureMPa, 3);
    Serial.println(" MPa");

    delay(750);
  }
}
